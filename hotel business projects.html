# ============================================================ # SETUP # ============================================================ import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns sns.set_style('whitegrid') plt.rcParams['figure.dpi'] = 100 CITY_COLOR, RESORT_COLOR = '#2E86AB', '#E67E22' PALETTE = {'City Hotel': CITY_COLOR, 'Resort Hotel': RESORT_COLOR} # ============================================================ # STAGE 1.1 — DATA OVERVIEW # ============================================================ df = pd.read_csv('hotel_bookings_data.csv') print(f"Rows: {df.shape[0]:,} Columns: {df.shape[1]}") print(f"Arrival years covered: {sorted(df['arrival_date_year'].unique())}") # ============================================================ # STAGE 1.2 — DATA ASSESSMENT: MISSING VALUES # ============================================================ missing = df.isna().sum() missing = missing[missing > 0].sort_values(ascending=False) print("Columns with missing values:") print(missing) print(f"\nAs % of rows:") print((missing / len(df) * 100).round(2)) df_clean = df.copy() df_clean['company'] = df_clean['company'].fillna(0) # 0 = no company on file df_clean['agent'] = df_clean['agent'].fillna(0) # 0 = booked without an agent df_clean['children'] = df_clean['children'].fillna(0) df_clean['city'] = df_clean['city'].fillna('Unknown') print("Remaining missing values:") print(df_clean.isna().sum().sum(), "total missing cells") # ---- Duplicates ---- dupes = df_clean.duplicated().sum() print(f"Duplicate rows found: {dupes:,} ({dupes/len(df_clean)*100:.1f}% of data)") df_clean = df_clean.drop_duplicates() print(f"Shape after removing duplicates: {df_clean.shape}") # ---- Inconsistent values: 'meal' ---- print("meal value counts:") print(df_clean['meal'].value_counts()) df_clean['meal'] = df_clean['meal'].replace({'Undefined': 'No Meal', 'SC': 'No Meal'}) print(df_clean['meal'].value_counts()) # ---- Anomalies: ADR and zero-guest bookings ---- print("ADR (average daily rate) summary:") print(df_clean['adr'].describe()) neg_adr = (df_clean['adr'] < 0).sum() extreme_adr = (df_clean['adr'] > 1000).sum() print(f"\nNegative ADR rows: {neg_adr}") print(f"Extreme ADR rows (>1000): {extreme_adr}") zero_guests = ((df_clean['adults'] + df_clean['children'] + df_clean['babies']) == 0).sum() print(f"Rows with zero total guests: {zero_guests}") before = len(df_clean) df_clean = df_clean[(df_clean['adr'] >= 0) & (df_clean['adr'] <= 1000)] df_clean = df_clean[(df_clean['adults'] + df_clean['children'] + df_clean['babies']) > 0] after = len(df_clean) print(f"Removed {before - after} anomalous rows") print(f"Final cleaned shape: {df_clean.shape}") # ---- Derived features used throughout Stage 2 ---- df_clean['total_stay'] = df_clean['stays_in_weekend_nights'] + df_clean['stays_in_weekdays_nights'] month_order = ['January','February','March','April','May','June','July', 'August','September','October','November','December'] df_clean['arrival_date_month'] = pd.Categorical( df_clean['arrival_date_month'], categories=month_order, ordered=True ) print("Cleaning complete. df_clean is ready for analysis.") # ============================================================ # STAGE 2.1 — BOOKING SHARE & SEASONALITY # ============================================================ share = df_clean['hotel'].value_counts() share_pct = df_clean['hotel'].value_counts(normalize=True) * 100 fig, ax = plt.subplots(figsize=(6,6)) ax.pie(share, labels=share.index, autopct='%1.1f%%', startangle=90, colors=[PALETTE[h] for h in share.index], wedgeprops={'edgecolor':'white','linewidth':2}) ax.set_title('Share of Bookings by Hotel Type', fontsize=14, fontweight='bold') plt.show() print(f"City Hotel: {share_pct['City Hotel']:.1f}% | Resort Hotel: {share_pct['Resort Hotel']:.1f}%") monthly = df_clean.groupby(['arrival_date_month','hotel'], observed=True).size().unstack() fig, ax = plt.subplots(figsize=(10,5)) for hotel in monthly.columns: ax.plot(monthly.index, monthly[hotel], marker='o', label=hotel, color=PALETTE[hotel], linewidth=2.5) ax.set_title('Bookings per Month by Hotel Type', fontsize=14, fontweight='bold') ax.set_xlabel('Month'); ax.set_ylabel('Number of Bookings') ax.legend(); ax.grid(alpha=0.3) plt.xticks(rotation=45) plt.tight_layout() plt.show() busiest = monthly.sum(axis=1).idxmax() quietest = monthly.sum(axis=1).idxmin() print(f"Busiest month overall: {busiest}") print(f"Quietest month overall: {quietest}") # ============================================================ # STAGE 2.2 — STAY DURATION vs CANCELLATION # ============================================================ cancel_rate_hotel = df_clean.groupby('hotel')['is_canceled'].mean() * 100 fig, ax = plt.subplots(figsize=(6,5)) bars = ax.bar(cancel_rate_hotel.index, cancel_rate_hotel.values, color=[PALETTE[h] for h in cancel_rate_hotel.index]) for bar in bars: h = bar.get_height() ax.text(bar.get_x()+bar.get_width()/2, h+0.5, f'{h:.1f}%', ha='center', fontweight='bold') ax.set_title('Cancellation Rate by Hotel Type', fontsize=14, fontweight='bold') ax.set_ylabel('Cancellation Rate (%)') plt.tight_layout() plt.show() print(cancel_rate_hotel.round(1)) df_clean['stay_bucket'] = df_clean['total_stay'].clip(upper=15) stay_cancel = (df_clean.groupby(['stay_bucket','hotel'], observed=True)['is_canceled'] .mean().mul(100).unstack()) fig, ax = plt.subplots(figsize=(10,5)) for hotel in stay_cancel.columns: ax.plot(stay_cancel.index, stay_cancel[hotel], marker='o', label=hotel, color=PALETTE[hotel], linewidth=2) ax.set_title('Cancellation Rate vs. Length of Stay', fontsize=14, fontweight='bold') ax.set_xlabel('Total Nights Stayed (15 = 15 or more)') ax.set_ylabel('Cancellation Rate (%)') ax.legend(); ax.grid(alpha=0.3) plt.tight_layout() plt.show() for hotel in ['City Hotel','Resort Hotel']: sub = df_clean[df_clean['hotel']==hotel] corr = sub['total_stay'].corr(sub['is_canceled']) print(f"{hotel}: correlation(total_stay, is_canceled) = {corr:.3f}") # ============================================================ # STAGE 2.3 — LEAD TIME vs CANCELLATION # ============================================================ df_clean['lead_time_bucket'] = pd.cut( df_clean['lead_time'], bins=[-1,7,30,60,90,180,365, df_clean['lead_time'].max()], labels=['0-7','8-30','31-60','61-90','91-180','181-365','365+'] ) lead_cancel = (df_clean.groupby(['lead_time_bucket','hotel'], observed=True)['is_canceled'] .mean().mul(100).unstack()) fig, ax = plt.subplots(figsize=(10,5)) x = np.arange(len(lead_cancel.index)); width = 0.35 for i, hotel in enumerate(lead_cancel.columns): ax.bar(x + i*width, lead_cancel[hotel], width, label=hotel, color=PALETTE[hotel]) ax.set_xticks(x + width/2); ax.set_xticklabels(lead_cancel.index) ax.set_title('Cancellation Rate vs. Lead Time', fontsize=14, fontweight='bold') ax.set_xlabel('Lead Time (days before arrival)') ax.set_ylabel('Cancellation Rate (%)') ax.legend(); ax.grid(alpha=0.3, axis='y') plt.tight_layout() plt.show() print(lead_cancel.round(1)) for hotel in ['City Hotel','Resort Hotel']: sub = df_clean[df_clean['hotel']==hotel] corr = sub['lead_time'].corr(sub['is_canceled']) print(f"{hotel}: correlation(lead_time, is_canceled) = {corr:.3f}")
